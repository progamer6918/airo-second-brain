   184	async function flushAutoMergeBuffer(
   185	  env: any,
   186	  ctx: ExecutionContext,
   187	  chatId: number,
   188	  userId: string,
   189	  reason: string
   190	): Promise<any | null> {
   191	  const key = getAutoMergeBufferKey(chatId, userId);
   192	  const entry = autoMergeBuffers.get(key);
   193	
   194	  if (!entry || entry.flushing) return null;
   195	
   196	  entry.flushing = true;
   197	  autoMergeBuffers.delete(key);
   198	
   199	  const mergedText = entry.parts.map((x) => x.trim()).filter(Boolean).join("\n").trim();
   200	  if (!mergedText) return null;
   201	
   202	  try {
   203	    const page = await createRecentCapture(env, mergedText);
   204	    const initialRoute = classifyInitialRoute(mergedText);
   205	    const job = runFinalRouting(env, mergedText, initialRoute, page.id, page.url)
   206	      .catch((err: any) => console.error("Final routing failed", err?.message || err));
   207	
   208	    ctx.waitUntil(job);
   209	
   210	    await sendTelegram(env, chatId, buildBubuReply(page.url));
   211	    return page;
   212	  } catch (err: any) {
   213	    console.error(`Auto merge flush failed: ${reason}`, err?.message || err);
   214	    await sendTelegram(
   215	      env,
   216	      chatId,
   217	      `❌ Gagal mencatat gabungan pesan ke Notion.\n\n${String(err.message || err).slice(0, 500)}`
   218	    ).catch((sendErr: any) => console.error("Failed to send auto merge error", sendErr?.message || sendErr));
   219	
   220	    return null;
   221	  }
   222	}

   224	function scheduleAutoMergeIdleFlush(
   225	  env: any,
   226	  ctx: ExecutionContext,
   227	  chatId: number,
   228	  userId: string,
   229	  idleVersion: number
   230	): void {
